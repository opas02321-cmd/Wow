local function checkpoint()
    if not uiAlive then error("Frekksec UI was unloaded", 0) end
    buildCount += 1
    if buildCount % 64 == 0 or os.clock() - lastYield > 0.006 then
        realTask.wait()
        lastYield = os.clock()
        if not uiAlive then error("Frekksec UI was unloaded", 0) end
    end
end

local function layoutSize(object)
    local scale, ancestor = 1, object
    while ancestor do
        local uiScale = ancestor:FindFirstChildOfClass("UIScale")
        if uiScale then scale *= uiScale.Scale end
        ancestor = ancestor.Parent
    end
    return object.AbsoluteSize / math.max(scale, 0.01)
end

local function softGlow(parent, color, thickness, transparency, skipShadow)
    local settings=create("Folder",{Name="LunarGlowSettings",Parent=parent})
    local glow=create("UIStroke",{
        Name="LunarSoftGlow",Color=color or Lunar.Theme.Accent,
        Thickness=math.clamp((thickness or 4)*0.5,1.5,4),
        Transparency=math.clamp(transparency or 0.72,0,1),
        Enabled=parent:IsA("TextButton") and not skipShadow,Parent=settings,
    })
    local shadow
    if glow.Enabled then
        local success,result=pcall(create,"UIShadow",{
            Name="LunarControlShadow",Color=glow.Color,
            BlurRadius=UDim.new(0,6),Offset=UDim2.fromOffset(0,1),
            Spread=UDim2.fromOffset(0,0),Transparency=1,Parent=parent,
        })
        if success then shadow=result end
    end
    local function refresh()
        if not parent.Parent or not shadow then return end
        local strength=Lunar.GlowStrength
        local intensity=math.clamp((1-glow.Transparency)*strength,0,1)
        shadow.Color=glow.Color
        shadow.BlurRadius=UDim.new(0,math.clamp(glow.Thickness*2.2,4,9))
        shadow.Spread=UDim2.fromOffset(0,0)
        shadow.Transparency=1-intensity*0.72
        shadow.Enabled=glow.Enabled and parent.Visible and intensity>0.05
    end
    local __array2=({"Color","Thickness","Transparency","Enabled"}); for _=1,#__array2 do local property=__array2[_]; if property==nil then break end
        glow:GetPropertyChangedSignal(property):Connect(refresh)
    end
    parent:GetPropertyChangedSignal("Visible"):Connect(refresh)
    glowControllers[glow]=refresh
    parent.Destroying:Connect(function() glowControllers[glow]=nil end)
    refresh()
    return glow
end

end

local function hexToColor(hex)
    hex = tostring(hex or ""):gsub("#", "")
    if #hex ~= 6 then return Lunar.Theme.Accent end
    local r = tonumber(hex:sub(1,2),16) or 255
    local g = tonumber(hex:sub(3,4),16) or 103

    return Color3.fromRGB(r,g,b)

local function ensureFolder(path)
    if type(isfolder) ~= "function" or type(makefolder) ~= "function" then return false end
    local built = ""
    for part in tostring(path):gmatch("[^/]+") do
        built = built == "" and part or (built .. "/" .. part)
        if not isfolder(built) then pcall(makefolder, built) end
    end
    return isfolder(path)
end

local function inputMatchesEnum(input, value)
    if typeof(value) ~= "EnumItem" then return false end
    if value.EnumType == Enum.KeyCode then
        return input.UserInputType == Enum.UserInputType.Keyboard and input.KeyCode == value
    end
    if value.EnumType == Enum.UserInputType then
        return input.UserInputType == value
    end
    return false
end
local function enumToString(value)
    if typeof(value) ~= "EnumItem" then return tostring(value) end
    if value == Enum.KeyCode.Unknown then return "None" end
    if value.EnumType == Enum.UserInputType then
        local aliases = {
            [Enum.UserInputType.MouseButton1] = "MB1",
            [Enum.UserInputType.MouseButton2] = "MB2",
            [Enum.UserInputType.MouseButton3] = "MB3",
        }
        return aliases[value] or value.Name
    end
    return value.Name
end
local function stringToEnum(value)
    if typeof(value) == "EnumItem" then return value end
    local text = tostring(value or "")
    local aliases = {
        MB1 = Enum.UserInputType.MouseButton1,
        MB2 = Enum.UserInputType.MouseButton2,
        MB3 = Enum.UserInputType.MouseButton3,
        MouseButton1 = Enum.UserInputType.MouseButton1,
        MouseButton2 = Enum.UserInputType.MouseButton2,
        MouseButton3 = Enum.UserInputType.MouseButton3,
        None = Enum.KeyCode.Unknown,
    }
    if aliases[text] then return aliases[text] end
    if Enum.KeyCode[text] then return Enum.KeyCode[text] end
    if Enum.UserInputType[text] then return Enum.UserInputType[text] end
    return Enum.KeyCode.Unknown
end
local function serializeValue(value)
    if typeof(value) == "Color3" then
        return {__type="Color3", R=value.R, G=value.G, B=value.B}
    elseif typeof(value) == "EnumItem" then
        return {__type="EnumItem", EnumType=tostring(value.EnumType), Name=value.Name}
    elseif type(value) == "table" then
        local out = {}
        for k,v in next, (value) do out[k] = serializeValue(v) end
        return out
    end
    return value
end
local function deserializeValue(value)
    if type(value) ~= "table" then return value end
    if value.__type == "Color3" then
        return Color3.new(tonumber(value.R) or 1, tonumber(value.G) or 1, tonumber(value.B) or 1)
    elseif value.__type == "EnumItem" then
        local enumName = tostring(value.EnumType or ""):match("Enum%.(.+)")
        local enumType = enumName and Enum[enumName]
        return enumType and enumType[value.Name] or nil
    end
    local out = {}
    for k,v in next, (value) do out[k] = deserializeValue(v) end
    return out
end

local function iconLine(parent, x1, y1, x2, y2, width, color)
    local dx,dy = x2-x1,y2-y1
    local len = math.sqrt(dx*dx+dy*dy)
    local f = create("Frame", {
        AnchorPoint = Vector2.new(0.5,0.5),
        Position = UDim2.fromOffset((x1+x2)/2,(y1+y2)/2),
        Size = UDim2.fromOffset(len,width or 2),
        Rotation = math.deg(math.atan2(dy,dx)),
        BackgroundColor3 = color,
        BorderSizePixel = 0,
        Parent = parent,
    })
    corner(f,99)
    return f
end

            Position=UDim2.fromOffset(x,y),Size=UDim2.fromOffset(w,h),

    local function surfaceContains(surface, point)
        if not surface.Parent then return false end
        local ancestor = surface
        while ancestor do
            if ancestor:IsA("GuiObject") and not ancestor.Visible then return false end
            if ancestor:IsA("ScreenGui") then
                if not ancestor.Enabled then return false end
                break
            end
            ancestor = ancestor.Parent
        end
        if not ancestor then return false end
        local p, size = surface.AbsolutePosition, surface.AbsoluteSize
        return point.X >= p.X and point.Y >= p.Y
            and point.X < p.X + size.X and point.Y < p.Y + size.Y
    end
    function self:RegisterInputSurface(surface)

    end
    function self:IsPointerOverMenu(point)
        if not gui.Enabled or not root.Visible then return false end
        point = point or getMenuMouseLocation()
        for surface in next, (inputSurfaces) do
            if surfaceContains(surface, point) then return true end
        end
        local dropdown = self._OpenDropdown
        local picker = self._OpenColorPicker
        return (dropdown ~= nil and dropdown:ContainsPoint(point))
            or (picker ~= nil and picker:ContainsPoint(point)) or false
    end
    local function pointerAction(_, state, input)
        local kind = input.UserInputType
        local key = kind == Enum.UserInputType.Touch and input or kind
        local point
        if kind == Enum.UserInputType.Touch then
            local inset = GuiService:GetGuiInset()
            point = Vector2.new(input.Position.X - inset.X, input.Position.Y - inset.Y)
        end
        local owned = pointerOwners[key]
        if state == Enum.UserInputState.Begin then
            owned = self:IsPointerOverMenu(point)
            pointerOwners[key] = owned
        elseif kind == Enum.UserInputType.MouseWheel then
            owned = self:IsPointerOverMenu(point)
        elseif owned == nil and state ~= Enum.UserInputState.End and state ~= Enum.UserInputState.Cancel then
            owned = self:IsPointerOverMenu(point)
        end
        if state == Enum.UserInputState.End or state == Enum.UserInputState.Cancel then
            pointerOwners[key] = nil
        end
        return owned and Enum.ContextActionResult.Sink or Enum.ContextActionResult.Pass
    end

function Window:_SelectTab(tab)
    if tab.ParentTab then
        self:_SelectTab(tab.ParentTab)
        tab.ParentTab:_SelectSubTab(tab)
        return
    end
    if self.ActiveTab == tab then return end
    if self._OpenDropdown then self._OpenDropdown:Close() end
    if self._OpenColorPicker then self._OpenColorPicker:Close() end
    if self.ActiveTab then
        self.ActiveTab.Page.Visible = false
        self.ActiveTab:_SetSelected(false)
    end
    self.ActiveTab = tab
    tab.Page.Visible = true
    tab:_SetSelected(true)
    self.CurrentLabel.Text = tab.Name
    self.SearchBox.Text = ""
    local isSettings = string.lower(tostring(tab.Name or "")):find("settings",1,true) ~= nil
    self._settingsSelected = isSettings
    if self.SettingsButton and self.SettingsStroke and self.SettingsGlow then
        tween(self.SettingsButton,0.14,{BackgroundTransparency=isSettings and 0.28 or 0.70})
        if self.SettingsLabel then tween(self.SettingsLabel,0.14,{TextColor3=isSettings and self.Theme.AccentSoft or self.Theme.Text}) end
        if self.SettingsScale then tween(self.SettingsScale,0.14,{Scale=isSettings and 1.012 or 1}) end
        if self.SettingsAccent then
            tween(self.SettingsAccent,0.16,{BackgroundTransparency=isSettings and 0 or 1,Size=UDim2.fromOffset(3,isSettings and 24 or 0)})
        end
        self.SettingsButton.ZIndex = isSettings and 35 or 20
        self.SettingsStroke.Transparency = isSettings and 0.05 or 1
        self.SettingsGlow.Transparency = isSettings and 0.40 or 0.95
        if self.SettingsIcon then self.SettingsIcon:SetColor(isSettings and self.Theme.AccentSoft or self.Theme.TextDim) end
    end
end

function Tab:_RefreshCanvas()
    if self._canvasRefreshQueued then return end
    self._canvasRefreshQueued = true
    task.defer(function()
        self._canvasRefreshQueued = false
        if not self.Page or not self.Page.Parent then return end
        local inset = self.ColumnPadding
        local extraHeight = inset.Top + inset.Bottom
        local leftHeight = math.max(0, self.LeftList.AbsoluteContentSize.Y + extraHeight)
        local rightHeight = math.max(0, self.RightList.AbsoluteContentSize.Y + extraHeight)
        self.Left.CanvasSize = UDim2.fromOffset(0,leftHeight)
        self.Right.CanvasSize = UDim2.fromOffset(0,rightHeight)
    end)
end
function Tab:AddGroupbox(info)
    if type(info)=="string" then info={Name=info,Side=1} end
    info=info or {}
    local name=info.Name or info.Text or ("Groupbox "..tostring(#self.Cards+1))

end

function Tab:_SelectSubTab(tab)
    if type(tab)=="string" then tab=self.SubtabByName and self.SubtabByName[tab] end
    if not tab or tab.ParentTab~=self or self.ActiveSubtab==tab then return end
    local window=self.Window
    if window._OpenDropdown then window._OpenDropdown:Close() end
    if window._OpenColorPicker then window._OpenColorPicker:Close() end
    if self.ActiveSubtab then
        self.ActiveSubtab.Page.Visible=false
        self.ActiveSubtab:_SetSelected(false)
    end
    self.ActiveSubtab=tab
    tab.Page.Visible=true
    tab:_SetSelected(true)
    if window.SearchQuery~="" or window.SearchBox.Text~="" then
        window.SearchQuery="";window.SearchBox.Text=""
        window:_ApplySearch()
    end
    tab:_LayoutColumns()
    tab:_RefreshCanvas()
end
function Tab:SelectSubTab(tab)
    self.Window:_SelectTab(self)
    self:_SelectSubTab(tab)
end

local function fitCompactControl(row, label, control, height)
    label.TextWrapped = true
    label.TextTruncate = Enum.TextTruncate.None
    local function fit()
        if not row.Parent then return end
        local width = layoutSize(row).X
        if width <= 24 then return end
        local caption = math.max(18, math.ceil(textBounds(label.Text, label.TextSize, label.Font, width - 20).Y))
        label.Position = UDim2.fromOffset(10, 4)
        label.Size = UDim2.new(1, -20, 0, caption)
        control.AnchorPoint = Vector2.new(0, 0)
        control.Position = UDim2.fromOffset(10, caption + 8)
        control.Size = UDim2.new(1, -20, 0, height)
        row.Size = UDim2.new(1, 0, 0, caption + height + 15)
    end
    row:GetPropertyChangedSignal("AbsoluteSize"):Connect(fit)
    label:GetPropertyChangedSignal("Text"):Connect(fit)
    task.defer(fit)
end

local function bindDropdownPopup(window, button, list, api, arrow)
    local connection
    local function contains(gui, point)
        local p, z = gui.AbsolutePosition, gui.AbsoluteSize
        return point.X >= p.X and point.X <= p.X + z.X and point.Y >= p.Y and point.Y <= p.Y + z.Y
    end
    function api:ContainsPoint(point)
        return contains(button, point) or (list.Visible and contains(list, point))
    end
    function api:Close()
        list.Visible = false
        arrow.Rotation = 0
        if connection then connection:Disconnect(); connection = nil end
        if window._OpenDropdown == api then window._OpenDropdown = nil end
    end
    local function position()
        if not button.Parent or not window.Gui.Enabled then api:Close(); return end
        local ancestor = button
        local center = button.AbsolutePosition + button.AbsoluteSize * 0.5
        while ancestor and ancestor ~= window.Gui do
            if ancestor:IsA("GuiObject") then
                if not ancestor.Visible or (ancestor.ClipsDescendants and not contains(ancestor, center)) then
                    api:Close(); return
                end
            end
            ancestor = ancestor.Parent
        end
        local layer = window.PopupLayer
        local origin, area = layer.AbsolutePosition, layer.AbsoluteSize
        local p, z = button.AbsolutePosition - origin, button.AbsoluteSize
        local width = math.min(math.max(z.X, 152), math.max(1, area.X - 16))
        local height = math.min(list.CanvasSize.Y.Offset, 240, math.max(1, area.Y - 16))
        local x = math.clamp(p.X + z.X - width, 8, math.max(8, area.X - width - 8))
        local y = p.Y + z.Y + 4
        if y + height > area.Y - 8 then y = p.Y - height - 4 end
        y = math.clamp(y, 8, math.max(8, area.Y - height - 8))
        list.Size = UDim2.fromOffset(width, height)
        list.Position = UDim2.fromOffset(x, y)
    end
    button.Activated:Connect(function()
        if list.Visible then api:Close(); return end
        if api.BuildRows then api:BuildRows() end
        if list.CanvasSize.Y.Offset <= 0 then return end
        if window._OpenDropdown then window._OpenDropdown:Close() end
        if window._OpenColorPicker then window._OpenColorPicker:Close() end
        list.CanvasPosition = Vector2.new(0,0)
        window._OpenDropdown = api
        list.Visible = true
        arrow.Rotation = 180
        position()
        if list.Visible then connection = connectUI(RunService.RenderStepped, position) end
    end)
end

local function sanitizeId(text)
    return tostring(text or "Option"):gsub("[^%w_%-%.]","_")
end
local function decorateOption(window, card, name, api, optionType, explicitId, noSave)
    if not api then return nil end
    local base = sanitizeId(explicitId or ((card.Tab and card.Tab.Name or "Tab").."."..(card.Title or "Card").."."..tostring(name)))
    local id = base
    local n = 2
    local registry = optionType == "Toggle" and window.Toggles or window.Options
    while registry[id] and registry[id] ~= api do
        id = base .. "_" .. tostring(n)
        n += 1
    end
    api.Id = id
    api.Type = optionType
    api.NoSave = noSave == true
    api._listeners = api._listeners or {}
    api.Value = api.Get and api:Get() or api.Value
    local rawSet = api.Set
    if rawSet and not api.__DecoratedSet then
        api.Set = function(selfApi, value, silent)
            rawSet(selfApi, value, silent)
            selfApi.Value = selfApi.Get and selfApi:Get() or value
            if not silent then
                do local __fhArray30 = (selfApi._listeners)
                for _ = 1, math.huge do
                    local listener = rawget(__fhArray30, _)
                    if listener == nil then break end
                    listener(selfApi.Value)
                end end
            end
            return selfApi
        end
        api.SetValue = api.Set
        api.__DecoratedSet = true
    elseif not api.SetValue and rawSet then
        api.SetValue = rawSet
    end
    function api:OnChanged(fn)
        if type(fn) == "function" then table.insert(self._listeners, fn) end
        return self
    end
    function api:RunChanged()
        self.Value = self.Get and self:Get() or self.Value
        do local __fhArray31 = (self._listeners or {})
        for _ = 1, math.huge do
            local listener = rawget(__fhArray31, _)
            if listener == nil then break end listener(self.Value) end end
        return self
    end
    function api:SetId(newId)
        newId = sanitizeId(newId)
        if newId == "" then return self end
        local currentRegistry = self.Type == "Toggle" and window.Toggles or window.Options
        if currentRegistry[self.Id] == self then currentRegistry[self.Id] = nil end
        self.Id = newId
        currentRegistry[newId] = self
        return self
    end
    function api:Serialize()
        if self.Type == "Keybind" or self.Type == "KeyPicker" then
            return {Key=serializeValue(self:Get()), Mode=self.GetMode and self:GetMode() or self.Mode, Active=self.IsActive and self:IsActive() or self.Toggled}
        end
        return serializeValue(self.Get and self:Get() or self.Value)
    end
    function api:Deserialize(value)
        if (self.Type == "Keybind" or self.Type == "KeyPicker") and type(value) == "table" then
            local key = deserializeValue(value.Key)
            if key and self.Set then self:Set(key, false) end
            if value.Mode and self.SetMode then self:SetMode(value.Mode, false) end
            return
        end
        if self.Set then self:Set(deserializeValue(value), false) end
    end
    registry[id] = api
    return api
end

function Window:SetTheme(themeData, name)
    if type(themeData) ~= "table" then return false end
    local changed = false
    for key, color in next, (themeData) do
        if typeof(color) == "Color3" and color ~= self.Theme[key] then
            changed = true
            break
        end
    end
    if not changed then
        self.ThemeName = name or self.ThemeName
        return true
    end
    local old = cloneTable(self.Theme)
    local merged = cloneTable(self.Theme)
    for k,v in next, (themeData) do if typeof(v)=="Color3" then merged[k]=v end end
    local map = {}
    for key,oldColor in next, (old) do
        if typeof(oldColor)=="Color3" and typeof(merged[key])=="Color3" then map[colorToHex(oldColor)] = merged[key] end
    end
    local function mapped(color) return map[colorToHex(color)] or color end
    local objects = self.Gui:GetDescendants()
    if self.NotificationGui then
        do local __fhArray33 = (self.NotificationGui:GetDescendants())
        for _ = 1, math.huge do
            local obj = rawget(__fhArray33, _)
            if obj == nil then break end objects[#objects + 1] = obj end end
    end
    do local __fhArray34 = (objects)
    for _ = 1, math.huge do
        local obj = rawget(__fhArray34, _)
        if obj == nil then break end
        if obj:IsA("UIStroke") then obj.Color = mapped(obj.Color)
        elseif obj:IsA("UIGradient") then
            local kp={}
            do local __fhArray35 = (obj.Color.Keypoints)
            for _ = 1, math.huge do
                local point = rawget(__fhArray35, _)
                if point == nil then break end table.insert(kp,ColorSequenceKeypoint.new(point.Time,mapped(point.Value))) end end
            pcall(function() obj.Color=ColorSequence.new(kp) end)
        elseif obj:IsA("GuiObject") then
            obj.BackgroundColor3 = mapped(obj.BackgroundColor3)
            obj.BorderColor3 = mapped(obj.BorderColor3)
            if obj:IsA("TextLabel") or obj:IsA("TextButton") or obj:IsA("TextBox") then
                obj.TextColor3 = mapped(obj.TextColor3)
            elseif obj:IsA("ImageLabel") or obj:IsA("ImageButton") or obj:IsA("ViewportFrame") then
                obj.ImageColor3 = mapped(obj.ImageColor3)
            elseif obj:IsA("ScrollingFrame") then
                obj.ScrollBarImageColor3 = mapped(obj.ScrollBarImageColor3)
            end
        end
    end end
    for k,v in next, (merged) do self.Theme[k]=v end
    self.ThemeName=name or self.ThemeName
    if self.Floral then self.Floral:Rebuild() end
    do local __fhArray36 = (self.Tabs)
    for _ = 1, math.huge do
        local tab = rawget(__fhArray36, _)
        if tab == nil then break end
        if tab.ParentTab then tab:_SetSelected(tab.ParentTab.ActiveSubtab==tab) end
    end end
    return true
end

    return self:SetTheme(preset,name)

            local modifiers=v[3] or v.Modifiers
            if key~=nil then decoratedSet(selfApi,stringToEnum(key),silent) end
            if modeValue and selfApi.SetMode then selfApi:SetMode(modeValue,silent) end
            if type(modifiers)=="table" then selfApi.Modifiers=modifiers end
        else
            decoratedSet(selfApi,stringToEnum(v),silent)
        end

        selfApi.Mode=selfApi:GetMode()
        selfApi.Toggled=selfApi:IsActive()

    end

        selfApi.Mode=selfApi:GetMode()
        selfApi.Toggled=selfApi:IsActive()
        return selfApi
    end
    function api:Update()
        if currentMode=="Always" then active=true
        elseif self.Toggled~=nil then active=value~=Enum.KeyCode.Unknown and self.Toggled==true end
        self.Value=enumToString(self:Get())
        self.Mode=self:GetMode()
        self.Toggled=self:IsActive()
        self._window:_UpdateKeybindList()
        return self
    end
    local decoratedSetActive=api.SetActive
    api.SetActive=function(selfApi,v,silent)
        decoratedSetActive(selfApi,v,silent)
        selfApi.Toggled=selfApi:IsActive()
        return selfApi
    end
    function api:GetState() return self:IsActive() end
    table.insert(self.Window.Keybinds,api)
    self.Window:_UpdateKeybindList()
    return api

        end

        if newObject then newObject.Parent=viewport end
        return self
    end
    function api:SetHeight(v)
        self.Height=math.max(40,tonumber(v) or self.Height)
        holder.Size=UDim2.new(1,0,0,self.Height+10)
        viewport.Size=UDim2.new(1,-20,0,self.Height)
        if self._card and self._card.Resize then self._card:Resize() end
        return self
    end
    function api:SetVisible(v)
        self.Visible=not not v
        holder.Visible=self.Visible
        return self
    end
    api._card=self
    self.Window.Options[id]=api
    return api
end

function Lunar:MakeDraggable(frame, cutoff)
    local start, position, active
    frame.Active = true
    frame.InputBegan:Connect(function(input)
        if input.UserInputType ~= Enum.UserInputType.MouseButton1 and input.UserInputType ~= Enum.UserInputType.Touch then return end
        if input.Position.Y - frame.AbsolutePosition.Y > (cutoff or 40) then return end
        start, position, active = input.Position, frame.Position, true
    end)
    connectUI(UserInputService.InputChanged, function(input)
        if not active or not frame.Parent then return end
        if input.UserInputType ~= Enum.UserInputType.MouseMovement and input.UserInputType ~= Enum.UserInputType.Touch then return end
        local delta = input.Position - start
        frame.Position = UDim2.new(position.X.Scale, position.X.Offset + delta.X, position.Y.Scale, position.Y.Offset + delta.Y)
    end)
    connectUI(UserInputService.InputEnded, function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then active = false end
    end)
end

local selectTab = Window._SelectTab
function Window:_SelectTab(tab)
    if Lunar.Unloaded or tab.Building then return end
    self.ShowRequest = (self.ShowRequest or 0) + 1
    local request = self.ShowRequest
    if tab.Builder and not tab.Built then
        tab.Building = true
        local ok, problem = pcall(tab.Builder)
        tab.Building = false
        if not ok then Lunar:OnUIError(problem); return end
        tab.Built = true
        if Lunar.Unloaded or request ~= self.ShowRequest then return end
    end
    selectTab(self, tab)
    tab:_LayoutColumns()
    tab:_RefreshCanvas()
end

local addLabel = Card.AddLabel
function Card:AddLabel(text, wrap)
    local api = addLabel(self, text, wrap)
    api.TextLabel = api.Label
    local row, label = api.Container, api.Label
    label.TextWrapped = true
    label.TextTruncate = Enum.TextTruncate.None
    label.TextYAlignment = Enum.TextYAlignment.Top
    label.Position = UDim2.fromOffset(10, 4)
    local sizing = false
    local function resize()
        if sizing or not row.Parent then return end
        sizing = true
        local scale = self.Window.Root:FindFirstChildOfClass("UIScale")
        local width = math.max(40, row.AbsoluteSize.X / (scale and scale.Scale or 1) - 20)
        local height = math.max(20, math.ceil(textBounds(label.Text, label.TextSize, label.Font, width).Y) + 4)
        if label.Size.Y.Scale == 0 then height = math.max(height, label.Size.Y.Offset) end
        row.AutomaticSize = Enum.AutomaticSize.None
        label.AutomaticSize = Enum.AutomaticSize.None
        label.Size = UDim2.new(1, -20, 0, height)
        row.Size = UDim2.new(1, 0, 0, height + 8)
        sizing = false
    end
    row:GetPropertyChangedSignal("AbsoluteSize"):Connect(resize)
    label:GetPropertyChangedSignal("Text"):Connect(resize)
    label:GetPropertyChangedSignal("Size"):Connect(resize)
    task.defer(resize)
    return api
end

function Library:AttachButtonGlow(button, inputTarget)
    local __array16=(self.ButtonGlows); for _=1,#__array16 do local existing=__array16[_]; if existing==nil then break end
        if existing.Button == button then return existing end;
    end;
    local entry = {Button=button, Hover=false, Pressed=false};
    local surface = inputTarget or button;
    local ok,shadow=pcall(self.Create,self,'UIShadow',{
        Name='FrekksecButtonShadow';
        BlurRadius=UDim.new(0,7);
        Color=self.AccentColor;
        Offset=UDim2.fromOffset(0,1);
        Spread=UDim2.fromOffset(0,0);
        Transparency=1;
        Parent=button;
    });
    if ok then
        entry.Shadow=shadow;
        self:AddToRegistry(shadow,{Color='AccentColor'});
    end;
    function entry:Refresh(animate)
        if Library.Unloaded or not self.Button.Parent or not self.Shadow then return end;
        local boost = Library.GlowHover and (self.Pressed and 1.5 or self.Hover and 1.25 or 1) or 1;
        local alpha=1-math.clamp(Library.ButtonGlowStrength * boost * 0.68,0,0.72);
        self.Shadow.Color=Library.AccentColor;
        self.Shadow.Enabled=Library.ButtonGlowStrength>0;
        if self.Tween then self.Tween:Cancel() end;
        if animate then
            self.Tween=TweenService:Create(self.Shadow,TweenInfo.new(0.16,Enum.EasingStyle.Quad,Enum.EasingDirection.Out),{Transparency=alpha});
            self.Tween:Play();
        else self.Shadow.Transparency=alpha end;
    end;
    surface.MouseEnter:Connect(function() entry.Hover=true; entry:Refresh(true) end);
    surface.MouseLeave:Connect(function() entry.Hover=false; entry.Pressed=false; entry:Refresh(true) end);
    surface.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            entry.Pressed=true; entry:Refresh(true);
        end;
    end);
    surface.InputEnded:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            entry.Pressed=false; entry:Refresh(true);
        end;
    end);
    table.insert(self.ButtonGlows, entry);
    entry:Refresh(false);
    return entry;
end;
